\begin{frame}{Mutual Information and the Information Plane}
  \begin{columns}[c,onlytextwidth]
    \column{0.47\textwidth}
      \centering
      {\color{Muted}\large A network forms a Markov chain}\par
      \vspace{0.25cm}
      \[
        X \longrightarrow T_1 \longrightarrow T_2
          \longrightarrow \cdots \longrightarrow T_n
      \]
      \vspace{0.2cm}
      \[
        \mutualinfo{X}{T_1}
        \ge \mutualinfo{X}{T_2}
        \ge \cdots
        \ge \mutualinfo{X}{T_n}
      \]
      {\color{Muted}\small Later layers cannot recover discarded input information.}
    \column{0.47\textwidth}
      {\color{Accent}\large $\mutualinfo{X}{T}$}\par
      Input information retained by representation $T$\par
      \vspace{0.45cm}
      {\color{Teal}\large $\mutualinfo{T}{Y}$}\par
      Label information available for prediction\par
      \vspace{0.45cm}
      Each layer is one point in the information plane. Training first increases label information, then compresses input information.
  \end{columns}
  \vspace{0.25cm}
  \centering
  {\color{Muted}\small For the paper's balanced binary target, $\mutualinfo{T}{Y}\le H(Y)\approx 1$ bit.}
\end{frame}

\begin{frame}{The Information Bottleneck Objective}
  \centering
  \vspace{0.05cm}
  \[
    \boxed{
      \min_T
      \left[
        \underbrace{\mutualinfo{X}{T}}_{\text{complexity}}
        - \beta\,
        \underbrace{\mutualinfo{T}{Y}}_{\text{fitness}}
      \right]
    }
  \]
  \vspace{0.35cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \centering
      {\color{Accent}\large Compress}\par
      \vspace{0.15cm}
      Discard details of $X$ that do not help predict $Y$.
    \column{0.46\textwidth}
      \centering
      {\color{Teal}\large Predict}\par
      \vspace{0.15cm}
      Retain the information in $T$ that remains useful for $Y$.
  \end{columns}
  \vspace{0.5cm}
  $\beta$ controls the value placed on prediction relative to compression.\par
  \vspace{0.15cm}
  {\color{Muted}\small Compression alone would produce a constant representation. Varying $\beta$ traces the Information Bottleneck curve.}
\end{frame}

\begin{frame}{Experimental setup}

\begin{itemize}
    \item Synthetic binary classification problem
    \item 12 binary inputs $\Rightarrow 2^{12} = 4096$ possible input patterns
    \item Binary target $Y \in \{0,1\}$ generated probabilistically from $X$
    \item Fully connected DNN, trained with SGD and cross-entropy
    \item No explicit regularization
\end{itemize}

\vspace{0.3cm}

For each hidden layer $T_i$, the authors compute

\[
I(X;T_i)
\qquad \text{and} \qquad
I(T_i;Y)
\]

\[
I(X;T_i) = \text{input information / complexity}
\]

\[
I(T_i;Y) = \text{label information / fitness}
\]

\end{frame}
