\begin{frame}{Mutual information}
  \begin{columns}[T,onlytextwidth]
    \column{0.36\textwidth}
      {\color{Accent}\large Definition}\par
      \vspace{0.12cm}
      \begin{equation}
        I(Y;T) = H(Y)-H(Y\mid T).
      \end{equation}
      Mutual information measures how much observing $T$ reduces uncertainty about $Y$.
    \column{0.60\textwidth}
      {\color{Teal}\large Distributional and KL-divergence view}\par
      \vspace{0.12cm}
      {\small
      \begin{equation}
        \begin{aligned}
          I(X;T)
          &= \sum_{x,t} p(x,t)
            \log\frac{p(x,t)}{p(x)p(t)} \\
          &= D_{\mathrm{KL}}\!\left(
            p(x,t)\,\middle\|\,p(x)p(t)
          \right).
        \end{aligned}
      \end{equation}}
      It compares the joint distribution with the product of the marginals. It is zero exactly when $X$ and $T$ are independent.
  \end{columns}
\end{frame}

\begin{frame}{The information plane}
  \centering
  {\small
    The dependency structure is
    $Y\!\longleftarrow\!X\!\longrightarrow\!T_1\!\longrightarrow\!\cdots
    \!\longrightarrow\!T_n\!\longrightarrow\!\hat{Y}$.}
  \vspace{-0.05cm}
  \begin{columns}[c,totalwidth=0.94\textwidth]
    \column{0.46\textwidth}
      \centering
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[x=0.72cm,y=0.72cm,>=Latex]
        \draw[->,thick,Ink] (0,0) -- (6.1,0)
          node[below right] {$\mutualinfo{X}{T}$};
        \draw[->,thick,Ink] (0,0) -- (0,4.6)
          node[above left] {$\mutualinfo{T}{Y}$};
        \draw[Accent,very thick]
          (0.35,0.25) .. controls (1.2,2.1) and (3.0,3.55) .. (5.55,4.0);
        \node[Accent,above] at (4.15,3.85) {IB curve};
        \fill[Teal] (4.7,3.82) circle (2.6pt);
        \node[Teal,below right] at (4.7,3.82) {$T$};
      \end{tikzpicture}
      }
    \column{0.48\textwidth}
      Each hidden layer is represented by
      \begin{equation}
        \left(I(X;T),\,I(T;Y)\right).
      \end{equation}
      Moving left means retaining fewer details about the input. Moving up means retaining more information about the target.
      \vspace{0.25cm}
      The paper reports a fast upward fitting phase followed by a slower leftward compression phase.
  \end{columns}
  \vspace{-0.15cm}
  {\small
  \begin{equation}
    I(X;Y) \ge I(T_1;Y) \ge I(T_2;Y) \ge \cdots \ge I(T_n;Y) \ge I(\hat{Y};Y).
  \end{equation}
  \vspace{-0.2cm}
  \begin{equation}
    H(X) \ge I(X;T_1) \ge I(X;T_2) \ge \cdots \ge I(X;T_n) \ge I(X;\hat{Y}).
  \end{equation}}
\end{frame}

\begin{frame}{The Information Bottleneck}
  \centering
  \vspace{0.05cm}
  \[
    \boxed{
      \min_T
      \left[
        \underbrace{\mutualinfo{X}{T}}_{\text{complexity}}
        - \beta\,
        \underbrace{\mutualinfo{T}{Y}}_{\text{fitness}}
      \right]
    }
  \]
  \vspace{0.35cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \centering
      {\color{Accent}\large Compress}\par
      \vspace{0.15cm}
      Discard details of $X$ that do not help predict $Y$.
    \column{0.46\textwidth}
      \centering
      {\color{Teal}\large Predict}\par
      \vspace{0.15cm}
      Retain the information in $T$ that remains useful for $Y$.
  \end{columns}
  \vspace{0.5cm}
  $\beta$ controls the value placed on prediction relative to compression.\par
  \vspace{0.15cm}
  {\color{Muted}\small Compression alone would produce a constant representation. Varying $\beta$ traces the Information Bottleneck curve.}
\end{frame}

\begin{frame}{The role of noise}
\begin{itemize}
    \item \textbf{Core problem}:
    MI is blind to complexity when the map is deterministic \\ 
    %MI is invariant to any invertible relabeling of X — this is why a deterministic f is invisible to $I(X;Y)$
    If Y is a deterministic function of X $(y=f(x))$, then $H(Y|X=x)= 0$ $\forall x$ \\ and:
    $$I(X;Y)=H(Y) - H(Y|X) \quad \Rightarrow \quad I(X;Y) = H(Y)$$  
    %%%%
    \item \textbf{Solution:} \\
    Replace the hard label with a soft, margin-dependent one using a sigmoidal function:
    $$ p(y=1\mid x) = \psi\big(w \cdot x), \qquad \psi(u) = \frac{1}{1+\exp(-\gamma u)}$$
    %where $\theta$ is chosen so $p(y=1)\approx 0.5$ and $\gamma$ tuned so $I(X;Y)\approx 0.99$ bits.
    \begin{itemize}
        \item Points near the boundary are genuinely ambiguous
        \item Points far away are confident
        \item So $I(X;Y)$ now depends on the margin distribution — i.e., on the rule's geometry.
    \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{Experimental setup}

\begin{itemize}
    \item Synthetic binary classification problem
    \item 12 binary inputs $\Rightarrow 2^{12} = 4096$ possible input patterns
    \item Binary target label $Y \in \{0,1\}$ soften with rule from last slide
    \item Fully connected DNN, trained with SGD and cross-entropy
    \item tanh as hidden activations and sigmoid as output
    \item No explicit regularization
\end{itemize}

\vspace{0.3cm}

For each hidden layer $T_i$, the authors compute
\[
I(X;T_i)
\qquad \text{and} \qquad
I(T_i;Y)
\]
by creating a \textbf{hard many-to-one mapping} (a quantizer):
\begin{itemize}
    \item Binning: 30 equal-width bins per neuron, over $[−1,1]$. Same trick as the label noise — without it $I(X;T_i)=H(X)=12 \text{bits}$ trivially, at every layer, every epoch.
\end{itemize}
 
\end{frame}
