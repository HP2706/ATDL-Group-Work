\begin{frame}{Training at scale}

\begin{itemize}
    \item 1M iterations of initial pre-training
    \item First 100k iterations is warmup
    \begin{itemize}
        \item learning rate increases linearly to $4\times10^{-4}$
        \item teacher temperature is also warmed up linearly
    \end{itemize}
    \item After warmup:
    \begin{itemize}
        \item constant learning rate
        \item constant teacher temperature
    \end{itemize}
    \item Replaces the cosine schedules used in DINOv2
    \item Global performance improves with longer training, while dense performance degrades after about 200k iterations
    \item Gram anchoring is introduced after 1M iterations to improve dense performance
\end{itemize}
\end{frame}

\begin{frame}{Pretraining losses}
  \begin{minipage}{\linewidth}
  \scriptsize
  \textbf{Prototype:} one learned row of the head's final weight matrix,
  not a class token or label. The DINO head scores the CLS summary against these rows;
  iBOT has a separate head for patches. $p_t(c)$ is the teacher's weight for vector $c$.\\
  \textbf{Collapse control:} Sinkhorn balances prototype use across a batch;
  KoLeo spreads CLS features. $p_t$ is the balanced target, $p_s$ the student prediction.
  \end{minipage}
  \vspace{-0.12cm}
  \footnotesize
  \begin{align*}
    \mathcal L_{\mathrm{DINO}}
      &= -\sum_c p^{\mathrm{CLS}}_{t}(c)\log p^{\mathrm{CLS}}_{s}(c)
      &&\text{match CLS prototype distributions across crops},\\[-2pt]
    \mathcal L_{\mathrm{iBOT}}
      &= -\frac{1}{|M|}\sum_{i\in M}\sum_c p_{t,i}(c)\log p_{s,i}(c)
      &&\text{match prototype distributions for masked patches }(M),\\[-2pt]
    \mathcal L_{\mathrm{DKoLeo}}
      &= -\frac{1}{B}\sum_{b=1}^{B}\log\!\left(\min_{j\ne b}
          \|z_b-z_j\|_2\right)
      &&\text{spread normalized CLS features }(z).
  \end{align*}
  \vspace{-0.16cm}
  \begin{center}
    \normalsize $\displaystyle
      \mathcal L_{\mathrm{Pre}}=\mathcal L_{\mathrm{DINO}}
      +\mathcal L_{\mathrm{iBOT}}+0.1\mathcal L_{\mathrm{DKoLeo}}$
  \end{center}
\end{frame}

\begin{frame}{Gram anchoring restores patch relationships}
  \small
  \begin{center}
    $\displaystyle
      \mathcal L_{\mathrm{Gram}}=
      \left\|X_SX_S^{\top}-X_GX_G^{\top}\right\|_F^2$
  \end{center}
  $X_S$ and $X_G$ are normalized student and earlier-teacher patch features.
  The new term preserves \textbf{patch-to-patch cosine similarities} while the
  original losses continue to train the student.

  \vspace{0.08cm}
  \centering
  \includegraphics[width=0.68\linewidth,trim=0 0 0 280,clip]{Presentation2/figures/dinov3-figure10.jpg}\par
  {\scriptsize Fig. 10: before (top) and after (bottom) high-resolution Gram refinement.}

  \vspace{0.07cm}
  {\footnotesize \textbf{Ablation (Fig. 9b):} ADE20k 50.3 $\to$ 55.7 mIoU;
  NYU depth 0.307 $\to$ 0.281 RMSE; ImageNet linear 88.2 $\to$ 88.0.}
\end{frame}

\begin{frame}{Critique and conclusion}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \cbox{shown}{Shown}{%
        \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\leftmargini}{8pt}
          \item Dense quality degrades while global keeps improving
          \item Dense quality restored almost "for free" globally with Gram
          \item ViT-H+ $\approx$ 7B teacher
          \item Recipe transfers to satellite
        \end{itemize}}
    \end{column}
    \begin{column}{0.32\textwidth}
      \cbox{hypo}{Hypothesised}{%
        \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\leftmargini}{8pt}
          \item Global loss dominates $\rightarrow$ patches align with CLS
          \item Early checkpoints hold better local structure
          \item SSL keeps scaling like LLMs
          \item Students don't degrade
        \end{itemize}}
    \end{column}
    \begin{column}{0.32\textwidth}
      \cbox{limit}{Limitations}{%
        \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\leftmargini}{8pt}
          \item Private data, curated with DINOv2
          \item Single-run ablations, short schedules
          \item OCR / zero-shot gap
          \item $\approx$2,600\,tCO$_2$eq project
        \end{itemize}}
    \end{column}
  \end{columns}
  \vfill
  \begin{center}
    \textbf{Main open question:} are the gains from Gram anchoring,\\
    or from 6$\times$ scale + private data?
  \end{center}
\end{frame}
